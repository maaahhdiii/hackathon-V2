    function showToast(message, ok = true) {
      toastEl.textContent = message;

      toastEl.style.display = 'block';
      setTimeout(() => { toastEl.style.display = 'none'; }, 2400);
    }

    async function fetchJson(url, options = {}) {
      const response = await fetch(url, options);
      const data = await response.json().catch(() => ({}));
      if (!response.ok || data.ok === false) {
        throw new Error(data.error || `HTTP ${response.status}`);
      }
      return data;
    }

    function hpColor(pct) {
      if (pct > 60) return '#53fca1';
      if (pct >= 30) return '#ffc857';
      return '#ff4d6d';
    }

    function fmtMMSS(sec) {
      const s = Math.max(0, Number(sec || 0));
      const m = String(Math.floor(s / 60)).padStart(2, '0');
      const r = String(s % 60).padStart(2, '0');
      return `${m}:${r}`;
    }

    async function startBattle() {
      try {
        await fetchJson('/api/battle/start', { method: 'POST' });
        showToast('Battle started');
      } catch (err) {
        showToast(err.message, false);
      }

    async function stopBattle() {
      try {
        await fetchJson('/api/battle/stop', { method: 'POST' });

    async function addBulkTeams() {
      try {
        const data = await fetchJson('/api/teams/add_bulk', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ count: 10, team_prefix: 'Team', register_mode: 'proxy_name' })
        });
        showToast(`Added ${data.registered}/${data.requested} teams`);
        refreshTeamsAndEvents();
      } catch (err) {
        showToast(err.message, false);
      }
    }

    async function addOneTeam() {
      const teamNo = Number(document.getElementById('teamNoInput').value);
      const teamName = document.getElementById('teamNameInput').value.trim();

      if (!Number.isInteger(teamNo) || teamNo < 1 || teamNo > 10) {
        showToast('Team number must be between 1 and 10', false);
        return;
      }

      try {
        const data = await fetchJson('/api/teams/add_one', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ team_no: teamNo, team_name: teamName })
        });
        const newIp = data?.team?.ip;
        if (newIp) openControls.add(newIp);
        showToast(`Added ${data?.team?.team_name || data?.team?.name || ('Team ' + teamNo)}`);
        await refreshTeamsAndEvents();
      } catch (err) {
        showToast(err.message, false);
      }
    }

    async function renameTeam(ip, oldName) {
      const name = prompt('New team name', oldName);
      if (!name) return;
      try {
        await fetchJson('/api/teams/rename', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ team_ip: ip, name })
        });
        showToast('Team renamed');
      } catch (err) {
        showToast(err.message, false);
      }
    }

    async function removeTeam(ip) {
      if (!confirm(`Remove team ${ip}?`)) return;
      try {
        await fetchJson(`/api/teams/${encodeURIComponent(ip)}`, { method: 'DELETE' });
        openControls.delete(ip);
        showToast('Team removed');
      } catch (err) {
        showToast(err.message, false);
      }
    }

    async function setHp(ip) {
      const service = document.getElementById(`svc-${ip}`).value;
      const hp = Number(document.getElementById(`hp-${ip}`).value);
      try {
        await fetchJson(`/api/teams/${encodeURIComponent(ip)}/hp`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ service, hp })
        });
        showToast('HP updated');
      } catch (err) {
        showToast(err.message, false);
      }
    }

        await fetchJson(`/api/teams/${encodeURIComponent(ip)}/score`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },

        });
        showToast('Score override set');
      } catch (err) {
        showToast(err.message, false);
      }
    }

    function renderTeams(teams) {
      const activeElement = document.activeElement;
      const activeId = activeElement && activeElement.id ? activeElement.id : '';
      const savedState = new Map();

      teams.forEach((team) => {
        const selectEl = document.getElementById(`svc-${team.ip}`);
        const hpEl = document.getElementById(`hp-${team.ip}`);
        const scoreEl = document.getElementById(`score-${team.ip}`);
        const ctrlEl = document.getElementById(`ctrl-${team.ip}`);

        savedState.set(team.ip, {
          open: openControls.has(team.ip) || (ctrlEl && ctrlEl.classList.contains('open')),
          service: selectEl ? selectEl.value : 'web',
          hp: hpEl ? hpEl.value : '',
          score: scoreEl ? scoreEl.value : ''
        });
      });

      teamsGrid.innerHTML = '';
      teams.forEach((team) => {
        const hp = team.hp || {};
        const scoreObj = team.score || {};
        const total = scoreObj.total ?? 0;

        const card = document.createElement('div');
        card.className = 'card';

        const hpRows = ['web', 'api', 'file', 'db'].map((svc) => {
          const val = hp[svc]?.current ?? 0;
          const max = hp[svc]?.max ?? (svc === 'web' ? 40 : 30);
          const pct = Math.max(0, Math.min(100, (val / max) * 100));
          return `
            <div class="hp-item">
              <span>${svc.toUpperCase()}</span>
              <div class="bar"><div class="fill" style="width:${pct}%;background:${hpColor(pct)}"></div></div>
              <span>${val}/${max}</span>
            </div>`;
        }).join('');

        card.innerHTML = `
          <div class="row">
            <span class="name" title="Rename" onclick="renameTeam('${team.ip}','${(team.name || '').replace(/'/g, "\\'")}')">${team.name}</span>
            <span class="ip">${team.ip}</span>
          </div>
          <div class="ip">Team ID: ${team.team_id ?? '-'} | Proxy: ${team.proxy_port ?? '-'} | IDE: ${team.ide_port ?? '-'}</div>
          <div class="score">${Number(total).toFixed(1)}</div>
          <div class="hp-group">${hpRows}</div>
          <button class="controls-toggle" type="button" onclick="toggleControls('${team.ip}')">Manual Controls</button>
          <div id="ctrl-${team.ip}" class="controls ${openControls.has(team.ip) ? 'open' : ''}">
            <div class="inline">
              <select id="svc-${team.ip}">
                <option value="web">web</option>
                <option value="api">api</option>
                <option value="file">file</option>
                <option value="db">db</option>
              </select>
              <input id="hp-${team.ip}" type="number" />
              <button class="small-btn" type="button" onclick="setHp('${team.ip}')">Set HP</button>
            </div>
            <div class="inline">
              <input id="score-${team.ip}" type="number" step="0.1" />

              <button class="small-btn remove" type="button" onclick="removeTeam('${team.ip}')">Remove Team</button>
            </div>
          </div>
        `;
        teamsGrid.appendChild(card);
      });

      savedState.forEach((state, ip) => {

        const hpEl = document.getElementById(`hp-${ip}`);
        const scoreEl = document.getElementById(`score-${ip}`);

        if (!ctrlEl || !selectEl || !hpEl || !scoreEl) return;

        if (state.open) {
          ctrlEl.classList.add('open');
          openControls.add(ip);

        if (state.service) selectEl.value = state.service;
        if (state.hp !== '') hpEl.value = state.hp;
        if (state.score !== '') scoreEl.value = state.score;

      if (activeId) {
        const activeAgain = document.getElementById(activeId);
        if (activeAgain && typeof activeAgain.focus === 'function') {
          activeAgain.focus();

    function toggleControls(ip) {
      const el = document.getElementById(`ctrl-${ip}`);
      if (!el) return;
      el.classList.toggle('open');
      if (el.classList.contains('open')) {
        openControls.add(ip);
      } else {
        openControls.delete(ip);
      }
    }

    function renderEvents(events) {
      eventsEl.innerHTML = '';
      if (!events.length) {
        eventsEl.innerHTML = '<div class="event">No events yet.</div>';
        return;
      }
      events.slice(-20).reverse().forEach((e) => {
        const line = document.createElement('div');

        function writeLog(msg) {

            const now = new Date().toLocaleTimeString();

            out.textContent = `[${now}] ${msg}


` + out.textContent;

        }


        function apiPath(path) {

            if (!path || typeof path !== 'string') return path;

            if (!path.startsWith('/')) return path;

            return apiBase + path;

        }


        function setVuln(v) {

            selectedVuln = v;

            selectedVulnLabel.textContent = v;

            ['sqli', 'xss', 'auth_bypass'].forEach((key) => {

                const el = document.getElementById('vbtn-' + key);

                if (el) el.classList.toggle('active', key === v);

            });

        }


        async function callApi(method, path, body) {

            const fullPath = apiPath(path);

            const opts = { method, headers: {} };

            if (body !== undefined) { opts.headers['Content-Type'] = 'application/json'; opts.body = JSON.stringify(body
); }

            try {

                const r = await fetch(fullPath, opts);

                const t = await r.text();

                writeLog(method + ' ' + fullPath + '

' + t);

                return { ok: r.ok, text: t };

            } catch (err) {

                writeLog(method + ' ' + fullPath + '

' + 'request error: ' + err.message);

                return { ok: false, text: String(err.message) };

            }

        }


        async function refreshHealth() {

            const r = await fetch(apiPath('/health'));

            const t = await r.text();

            writeLog('GET /health

' + t);

            try {

                const d = JSON.parse(t);

                healthChip.textContent = `Health: ${d.status || '--'} | HP ${d.hp ?? '--'}/${d.max_hp ?? '--'}`;

                healthChip.className = 'chip ' + ((d.status === 'online' || d.status === 'degraded') ? 'ok' : '');

            } catch (_) {}

        }

            const res = await callApi('POST', '/login', { username: val('username'), password: val('password') });

            try { const data = JSON.parse(res.text); if (data.token) document.getElementById('token').value = data.token
; } catch (_) {}

        }

        async function profile() {

            const token = val('token');

            const r = await fetch(apiPath('/profile'), { headers: { Authorization: 'Bearer ' + token } });

            writeLog('GET /profile

' + await r.text());

        }

    function fileExt(name) {
      const i = name.lastIndexOf('.');
      return i >= 0 ? name.slice(i + 1).toLowerCase() : 'py';
    }

    function monacoLangForFile(name) {
      return extToMonaco[fileExt(name)] || 'python';
    }

    async function api(url, opts = {}) {
      const res = await fetch(url, opts);
      const data = await res.json().catch(() => ({}));
      if (!res.ok) throw new Error(data.error || `HTTP ${res.status}`);
      return data;
    }

    function renderFiles() {
      const list = document.getElementById('fileList');
      list.innerHTML = '';
      files.forEach((f) => {
        const div = document.createElement('div');
        div.className = `file-item ${f === currentFile ? 'active' : ''}`;
        div.textContent = f;
        div.onclick = () => loadFile(f);
        list.appendChild(div);
      });
    }

    async function refreshFiles() {
      const data = await api('/api/files');
      files = data.files || [];
      if (!currentFile && files.length) currentFile = files[0];
      renderFiles();
      if (currentFile) await loadFile(currentFile);
    }

    async function loadFile(name) {
      const data = await api(`/api/files/${encodeURIComponent(name)}`);
      currentFile = name;
      model.setValue(data.content || '');
      monaco.editor.setModelLanguage(model, monacoLangForFile(name));
      const lang = monacoLangForFile(name);
      const sel = document.getElementById('langSelect');
      if ([...sel.options].some((o) => o.value === lang)) {
        sel.value = lang;
      }
      renderFiles();
    }

    function targetFileFor(tab, lang) {

      return `${tab}.${ext}`;
    }

    async function ensureActiveTabFile() {

      const target = targetFileFor(currentTab, lang);

      if (!files.includes(target)) {
        await api(`/api/files/${encodeURIComponent(target)}`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ content: '' })
        });
        files.push(target);
        files.sort();
      }

      if (currentFile !== target) {
        await loadFile(target);
      } else {
        renderFiles();
      }
    }

        body: JSON.stringify({ content: model.getValue() })
      });

      }).join('');

      logEl.scrollTop = logEl.scrollHeight;
